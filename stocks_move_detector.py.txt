# Sudden Stock Move Detector + News Explainer

**What this does:** scans a watchlist of tickers, flags any that just moved more than a threshold you set, and pulls recent headlines for each flagged name so you can see *why* it likely moved.

**Two honest limits, upfront:**
1. Free Yahoo Finance data (via `yfinance`) is delayed **~15 minutes**. That's not tick-by-tick real time — true real-time needs a paid feed (Polygon.io, IEX Cloud, or a broker API like Alpaca / Interactive Brokers). This tool is "fast" in the sense of *frequent, automatic checking*, not sub-second.
2. This notebook only runs when *you* run it — it doesn't sit in the background by itself. See the last section for how to make it run on a schedule.

Everything that counts as a "sudden move" is a plain, visible number in the CONFIG cell below — no hidden model, no black box. Change the numbers, the definition changes.
"""

# ============ CONFIG — Exhaustive Watchlist + Fundamental Filters ============
import pandas as pd
import yfinance as yf
import requests
import io

def get_exhaustive_watchlist():
    tickers = ["LCID"]
    headers = {'User-Agent': 'Mozilla/5.0'}
    try:
        us_url = "https://raw.githubusercontent.com/rreichel3/US-Stock-Symbols/main/all/all_tickers.txt"
        all_us_text = requests.get(us_url).text
        all_us = [t.strip().upper() for t in all_us_text.split('\n') if t.strip()]
        tickers.extend(all_us)
    except: pass
    try:
        ftse_url = 'https://en.wikipedia.org/wiki/FTSE_100_Index'
        ftse = pd.read_html(requests.get(ftse_url, headers=headers).text)[4]['EPIC'].tolist()
        tickers.extend([f"{t}.L" for t in ftse])
    except: pass
    tickers.extend(["^FTSE", "^N225", "DIA", "QQQ", "SPY"])
    return list(set(tickers))

WATCHLIST = get_exhaustive_watchlist()

# Fundamental Thresholds - Removed as per user request
# MIN_MARKET_CAP = 10_000_000
# MIN_PE_RATIO = 2
# MIN_EV_EBITDA = 2

# Technical Thresholds - NEW CRITERIA
DAILY_STD_DEV_MULTIPLIER = 3 # Daily move > 3 standard deviations
VOLUME_AVG_MULTIPLIER = 2    # Today's volume > 2x average volume
HISTORY_PERIOD = "1y"        # Period to fetch historical data for std dev and avg volume
INTRADAY_INTERVAL = "5m"
NEWS_PER_TICKER = 3
# ================================================================================

"""## Step 1 — scan the watchlist for sudden moves

Two separate checks, both visible above: a same-day move vs. yesterday's close, and a short-window move (catches something that just happened, not just a slow drift over the whole day).
"""

def scan_watchlist(tickers=WATCHLIST):
    rows = []
    print(f"Scanning {len(tickers)} tickers for daily move > {DAILY_STD_DEV_MULTIPLIER} std dev AND volume > {VOLUME_AVG_MULTIPLIER}x average...")
    for tkr in tickers:
        try:
            t = yf.Ticker(tkr)
            info = t.info

            # --- Fundamental Filtering ---
            # Removed: mcap, pe, ev_ebitda checks as per user request.

            # --- Price Action and Volume Scanning ---
            # Fetch historical data for calculating std dev and average volume
            hist_data = t.history(period=HISTORY_PERIOD)
            if hist_data.empty: continue

            # Calculate daily returns from historical data
            hist_data['Daily_Return'] = hist_data['Close'].pct_change()
            std_dev_returns = hist_data['Daily_Return'].std()

            # Calculate average volume from historical data (excluding current day if partial)
            avg_volume = hist_data['Volume'].mean()

            # Get current day's data (intraday) and current volume
            current_intraday_data = t.history(period="1d", interval=INTRADAY_INTERVAL)
            if current_intraday_data.empty: continue

            # Determine previous_close using info.get() and fallback to intraday data
            prev_close = info.get("previousClose")
            if prev_close is None:
                prev_close = float(current_intraday_data["Close"].iloc[0])

            last_price = current_intraday_data["Close"].iloc[-1]
            daily_pct = (last_price - prev_close) / prev_close
            current_volume = current_intraday_data["Volume"].sum() # Sum intraday volumes for daily total

            # Apply new conditions: daily move > N std dev AND current volume > M x average volume
            if (std_dev_returns > 0 and abs(daily_pct) >= DAILY_STD_DEV_MULTIPLIER * std_dev_returns) and \
               (avg_volume > 0 and current_volume >= VOLUME_AVG_MULTIPLIER * avg_volume):

                rows.append({
                    "ticker": tkr,
                    "mkt_cap": f"{info.get('marketCap', 0):,.0f}",
                    "pe_ratio": round(info.get('trailingPE', 0), 2),
                    "ev_ebitda": round(info.get('enterpriseToEbitda', 0), 2),
                    "last_price": round(float(last_price), 2),
                    "pct_daily": round(float(daily_pct) * 100, 2),
                    "current_volume": f"{current_volume:,.0f}",
                    "avg_volume": f"{avg_volume:,.0f}",
                    "std_dev_daily_return": round(std_dev_returns * 100, 2),
                    "flag": "volatility_volume"
                })
        except Exception as e:
            # print(f"Error processing {tkr}: {e}") # Uncomment for debugging
            continue
    return pd.DataFrame(rows)

"""## Step 2 — find out why

Two sources, used together so one going down doesn't leave you with nothing: Google News (free, no key, very reliable) and Yahoo Finance's own ticker news (handles both the old and new response formats yfinance has used).
"""

def get_news_google(query, max_results=NEWS_PER_TICKER):
    """Free, no API key needed. Recent Google News headlines matching `query`."""
    url = f"https://news.google.com/rss/search?q={urllib.parse.quote(query)}&hl=en-US&gl=US&ceid=US:en"
    feed = feedparser.parse(url)
    articles = []
    for entry in feed.entries[:max_results]:
        articles.append({
            "title": entry.get("title"),
            "link": entry.get("link"),
            "published": entry.get("published"),
            "source": "Google News",
        })
    return articles


def get_news_yfinance(ticker, max_results=NEWS_PER_TICKER):
    """
    Supplementary source straight from Yahoo Finance. yfinance's news schema has
    changed across versions (flat vs. nested under 'content'), so this handles both.
    """
    articles = []
    try:
        raw = yf.Ticker(ticker).news or []
        for item in raw[:max_results]:
            content = item.get("content", item)  # nested in newer versions, flat in older
            link = None
            if isinstance(content.get("clickThroughUrl"), dict):
                link = content["clickThroughUrl"].get("url")
            link = link or content.get("link")

            articles.append({
                "title": content.get("title"),
                "link": link,
                "published": content.get("pubDate") or item.get("providerPublishTime"),
                "source": "Yahoo Finance",
            })
    except Exception as e:
        print(f"[{ticker}] yfinance news error: {e}")
    return articles

"""## Step 3 — put it together into one report

For every flagged ticker: print the numbers that triggered the flag, the headlines, and a same-day price chart so you can see the move.
"""

def build_report(tickers=WATCHLIST):
    flagged = scan_watchlist(tickers)

    if flagged.empty:
        print("No stocks crossed the thresholds right now. Nothing to report.")
        return flagged

    print(f"=== {len(flagged)} stock(s) flagged at {datetime.now().strftime('%Y-%m-%d %H:%M:%S')} ===\n")

    for _, row in flagged.iterrows():
        tkr = row["ticker"]
        print(f"--- {tkr} ---")
        print(row.to_dict())

        articles = get_news_google(tkr) or get_news_yfinance(tkr)
        if not articles:
            print("  (no recent articles found for this ticker)")
        for a in articles:
            print(f"  • {a['title']}  [{a['source']}]")
            print(f"    {a['link']}")

        hist = yf.Ticker(tkr).history(period="1d", interval=INTRADAY_INTERVAL)
        plt.figure(figsize=(8, 3))
        plt.plot(hist.index, hist["Close"])
        plt.title(f"{tkr} — today")
        plt.xlabel("Time")
        plt.ylabel("Price")
        plt.grid(True)
        plt.tight_layout()
        plt.show()
        print()

    return flagged

"""## Run it"""

# Re-running the scan with fundamentals applied
report = build_report(WATCHLIST)
